for (var wxy_i = 1; wxy_i <= 9; wxy_i++) { for (var wxy_j = 1; wxy_j <= wxy_i; wxy_j++) { document.write(wxy_j + "×" + wxy_i + "=" + (wxy_j * wxy_i) + " "); } }
